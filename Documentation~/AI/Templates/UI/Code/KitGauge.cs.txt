// UI 템플릿 — 게이지(슬라이더): 체력 · 마나 · 경험치 · 로딩 · 버프 지속처럼 0~1을 채우는 것은 모두 이 하나로(개발자 2026-10-15 — 이것이 경우의 수 전부).
//   kind  : Line 직선 · Orb 원형 패널에 아래에서 차오름 · Ring 원 테두리를 12시부터 시계 방향으로 360° · Units 도형 나열(한 칸이 다 차야 다음 칸).
//           직선에 눈금만 치려면 segments.
//   shape : Rect 사각형 · Slant 평행사변형(차오르는 앞면도 같은 기울기) · Diamond 마름모(도형 나열만). 둥근 모양은 쓰지 않는다 — 모서리만 아주 약간(radius).
//   frame : Flush 게이지가 뒤 판을 꽉 채움 · Inset 단색 뒤 판이 게이지보다 inset(px)만큼 더 크게 둘러쌈(별도 테가 아니다).
//   색    : UXML 속성 color(게이지) · end-color(그라데이션 끝) · track-color(뒤 판) · loss-color · gain-color — 비우면 USS(--gauge-fill · --gauge-fill-end ·
//           --gauge-track · --gauge-loss · --gauge-gain) → 기본. 뒷 게이지 색은 비우면 게이지 색에서 만든다(줄 때 = 더 밝게, 찰 때 = 더 어둡게).
//           그라데이션(gradient): Along = 0~100 자리에 색을 고정(차는 만큼 드러남) · ByValue = 게이지 전체가 지금 값의 색(예: 체력이 줄면 초록 → 빨강).
//           색 지점은 color → end-color 둘, 또는 stops="0:#C86058, 50:#E0C060, 100:#78B070"(0~100 또는 0~1), 코드에서는 Gradient(Unity Gradient).
//           glow = 빛 번짐(지금 값 끝의 색으로).
//   움직임: 메인 게이지 뒤에 뒷 게이지 하나.
//           줄 때 — Drop(값): 메인이 바로 깎이고 밝은 뒷 게이지가 남는다 → Follow(): 뒷 게이지가 따라 준다(Drop의 follow가 참이면 followDelay 뒤 저절로).
//           찰 때 — Raise(값): 어두운 뒷 게이지가 먼저 뜬다 → FillUp(): 메인이 차오른다(Raise의 fill이 참이면 fillDelay 뒤 저절로).
//           Set(값)은 둘을 저절로 이어 부르는 지름길. Raise(값, fill: false)로 찰 몫을 미리 보여 두고 원할 때 FillUp — 단계마다 코드에 따로 잇는다.
//           Preview(음수)는 잃을 몫을 메인 위에 표시(얻을 몫의 미리보기는 Raise(값, fill: false)).
// UXML: <MomoiProject.KitGauge kind="Line" shape="Slant" frame="Inset" value="0.7" color="#C86058" class="kit-gauge" />
// 코드: gauge.Set(hp / (float)maxHp); · gauge.Raise(next, fill: false); … gauge.FillUp(); · gauge.Text = $"{hp} / {maxHp}";

        // 미리보기(Raise(값, fill: false))로 띄워 둔 찰 몫이 있는가 · 뒷 게이지의 끝.
        public bool HasPendingGain => _backKind == Back.Gain && _back > _main && !_filling;

        public float preview => _preview;
        // 그라데이션 색 지점(코드에서 Unity Gradient로 — 넣으면 color · end-color · stops보다 먼저).

        Color Fill => _color.a > 0f ? _color : _ussFill ?? new Color(0.78f, 0.38f, 0.35f);
        Color FillEnd => _endColor.a > 0f ? _endColor : _ussFillEnd ?? Color.Lerp(Fill, Color.white, 0.35f);
        Color Track => _trackColor.a > 0f ? _trackColor : _ussTrack ?? new Color(0.094f, 0.098f, 0.114f);
        Color Current => _gradient == Coloring.None ? Fill : Colors().Evaluate(_main);   // 지금 값 끝의 색
        Color Loss => _lossColor.a > 0f ? _lossColor : _ussLoss ?? Color.Lerp(Current, Color.white, 0.55f);    // 메인보다 밝게
        Color Gain => _gainColor.a > 0f ? _gainColor : _ussGain ?? Color.Lerp(Current, Color.black, 0.45f);    // 메인보다 어둡게

        public KitGauge()
        {
            AddToClassList("kit-gauge");
            generateVisualContent += DrawTrack;
            _backLayer = Layer("kit-gauge-layer--back", DrawBack);
            _glowLayer = Layer("kit-gauge-layer--glow", DrawGlow);   // 빛 번짐은 단색 층에(필터는 그라데이션 칠하기와 함께 그려지지 않는다)
            _fillLayer = Layer("kit-gauge-layer--fill", DrawFill);
            _overLayer = Layer("kit-gauge-layer--over", DrawOver);
            _label = new Label { pickingMode = PickingMode.Ignore };
            _label.AddToClassList("kit-gauge-text");
            _label.style.display = DisplayStyle.None;
            Add(_label);
            RegisterCallback<CustomStyleResolvedEvent>(e =>
            {
                var s = e.customStyle;
                _ussFill = s.TryGetValue(FillProperty, out var c) ? c : (Color?)null;
                _ussFillEnd = s.TryGetValue(FillEndProperty, out c) ? c : (Color?)null;
                _ussTrack = s.TryGetValue(TrackProperty, out c) ? c : (Color?)null;
                _ussLoss = s.TryGetValue(LossProperty, out c) ? c : (Color?)null;
                _ussGain = s.TryGetValue(GainProperty, out c) ? c : (Color?)null;
                if (s.TryGetValue(TickProperty, out c)) _tickColor = c;
                if (s.TryGetValue(PreviewProperty, out c)) _previewColor = c;
                _built = null;
                Glow();
                Repaint();
            });
            RegisterCallback<DetachFromPanelEvent>(_ => _tick?.Pause());
            Classes();
        }

        // ── 움직임: 단계마다 따로 부를 수 있다 ──

        // 지름길: 줄면 Drop(저절로 Follow), 늘면 Raise(저절로 FillUp). animate가 거짓이면 바로.
        public void Set(float target, bool animate = true)
        {
            target = Mathf.Clamp01(target);
            if (!animate || panel == null) { value = target; return; }
            if (target < _main) Drop(target);
            else if (target > _main) Raise(target);
            else if (_backKind == Back.Gain) { _back = _main; _backKind = Back.None; _filling = false; _fillWait = -1f; Repaint(); }
        }

        // 줄 때 ①: 메인을 바로 깎고, 밝은 뒷 게이지를 깎이기 전 자리에 남긴다. follow가 참이면 followDelay 뒤 저절로 Follow.
        public void Drop(float target, bool follow = true)
        {
            target = Mathf.Clamp01(target);
            if (target >= _main) return;
            var from = _backKind == Back.Loss ? Mathf.Max(_back, _main) : _main;
            _main = target;
            _back = from;
            _backKind = Back.Loss;
            _filling = false;
            _fillWait = -1f;
            _following = false;
            _followWait = follow ? followDelay : -1f;
            Run();
        }

        // 줄 때 ②: 밝은 뒷 게이지가 메인까지 따라 준다(followSeconds 동안).
        public void Follow()
        {
            if (_backKind != Back.Loss || _back <= _main) return;
            _followWait = -1f;
            _following = true;
            Run();
        }

        // 찰 때 ①: 어두운 뒷 게이지를 먼저 target까지 띄운다. fill이 참이면 fillDelay 뒤 저절로 FillUp, 거짓이면 FillUp을 부를 때까지 미리보기로 남는다.
        public void Raise(float target, bool fill = true)
        {
            target = Mathf.Clamp01(target);
            if (target <= _main) return;
            _back = target;
            _backKind = Back.Gain;
            _following = false;
            _followWait = -1f;
            _filling = false;
            _fillWait = fill ? fillDelay : -1f;
            Run();
        }

        // 찰 때 ②: 메인이 어두운 뒷 게이지까지 차오른다(fillSeconds 동안).
        public void FillUp()
        {
            if (_backKind != Back.Gain || _back <= _main) return;
            _fillWait = -1f;
            _filling = true;
            Run();
        }

        // 잃을 몫을 메인 위에 표시(-1~0, 0 = 지움). 얻을 몫의 미리보기는 Raise(값, fill: false).
        public void Preview(float lossAmount)
        {
            _preview = Mathf.Clamp(lossAmount, -1f, 0f);
            Repaint();
        }

        void Run()
        {
            Repaint();
            _lastTime = Time.realtimeSinceStartupAsDouble;
            _tick ??= schedule.Execute(Step).Every(16);
            _tick.Resume();
        }

        void Step()
        {
            var now = Time.realtimeSinceStartupAsDouble;
            var dt = (float)(now - _lastTime);
            _lastTime = now;
            if (_followWait >= 0f) { _followWait -= dt; if (_followWait < 0f) { _followWait = -1f; _following = true; } }
            if (_fillWait >= 0f) { _fillWait -= dt; if (_fillWait < 0f) { _fillWait = -1f; _filling = true; } }
            if (_following)
            {
                _back = Mathf.Max(_main, _back - dt / Mathf.Max(followSeconds, 0.01f));
                if (_back <= _main) { _following = false; _backKind = Back.None; }
            }
            if (_filling)
            {
                _main = Mathf.Min(_back, _main + dt / Mathf.Max(fillSeconds, 0.01f));
                if (_main >= _back) { _filling = false; _backKind = Back.None; }
            }
            Repaint();
            if (!_following && !_filling && _followWait < 0f && _fillWait < 0f) _tick.Pause();
        }

        // ── 그리기: 뒤 판(이 요소) → 뒷 게이지 → 메인 게이지(그라데이션 · 빛 번짐) → 잃을 몫 표시 · 눈금 ──

        VisualElement Layer(string className, System.Action<MeshGenerationContext> draw)
        {
            var layer = new VisualElement { pickingMode = PickingMode.Ignore };
            layer.AddToClassList("kit-gauge-layer");
            layer.AddToClassList(className);
            layer.style.position = Position.Absolute;
            layer.style.left = layer.style.top = layer.style.right = layer.style.bottom = 0;
            layer.generateVisualContent += draw;
            Add(layer);
            return layer;
        }

        void Classes()
        {
            EnableInClassList("kit-gauge--orb", _kind == Kind.Orb);
            EnableInClassList("kit-gauge--ring", _kind == Kind.Ring);
            EnableInClassList("kit-gauge--units", _kind == Kind.Units);
        }

        void Repaint()
        {
            MarkDirtyRepaint();
            _backLayer?.MarkDirtyRepaint();
            _glowLayer?.MarkDirtyRepaint();
            _fillLayer?.MarkDirtyRepaint();
            _overLayer?.MarkDirtyRepaint();
            if (_glow && _gradient != Coloring.None) Glow();   // 값에 따라 빛 색이 바뀐다
        }

        // 빛 번짐: 메인 게이지 아래의 단색 층에만, 지금 값 끝의 색으로(색이 바뀔 때만 다시 건다).
        void Glow()
        {
            if (_glowLayer == null) return;
            if (!_glow) { _glowLayer.style.filter = StyleKeyword.Null; _glowColor = Color.clear; return; }
            var light = Current;
            if (_glowColor.a > 0f && Mathf.Abs(light.r - _glowColor.r) + Mathf.Abs(light.g - _glowColor.g) + Mathf.Abs(light.b - _glowColor.b) < 0.01f) return;
            _glowColor = light;
            var shadow = new FilterFunction(FilterFunctionType.DropShadow);
            shadow.AddParameter(new FilterParameter(0f));
            shadow.AddParameter(new FilterParameter(0f));
            shadow.AddParameter(new FilterParameter(glowSize));
            shadow.AddParameter(new FilterParameter(new Color(light.r, light.g, light.b, 0.9f)));
            _glowLayer.style.filter = new List<FilterFunction> { shadow };
        }

        void DrawTrack(MeshGenerationContext context)
        {
            if (!Ready(out var size)) return;
            var p = context.painter2D;
            var c = size / 2f;
            var r = Mathf.Min(size.x, size.y) / 2f;
            switch (_kind)
            {
                case Kind.Ring: Stroke(p, c, r - thickness / 2f, thickness, 0f, 1f, Track); break;
                case Kind.Orb: Disc(p, c, r, Track); break;
                default:
                    foreach (var box in Boxes(size)) Paint(p, ShapeOf(box, size, 0f), Track);
                    break;
            }
        }

        void DrawBack(MeshGenerationContext context)
        {
            if (!Ready(out var size) || _back <= _main) return;
            Range(context.painter2D, size, _main, _back, _backKind == Back.Gain ? Gain : Loss);
        }

        void DrawGlow(MeshGenerationContext context)
        {
            if (!_glow || !Ready(out var size) || _main <= 0f) return;
            Range(context.painter2D, size, 0f, _main, Current);
        }

        void DrawFill(MeshGenerationContext context)
        {
            if (!Ready(out var size) || _main <= 0f) return;
            var p = context.painter2D;
            if (_gradient == Coloring.Along)
            {
                if (_kind == Kind.Ring) { RingAlong(p, size, 0f, _main); return; }
                UseGradient(p, size);
            }
            // 자리 그라데이션이면 칠하기 색을 건드리지 않는다(fillColor를 넣으면 fillGradient가 지워진다).
            Range(p, size, 0f, _main, _gradient == Coloring.Along ? (Color?)null : _gradient == Coloring.ByValue ? Current : Fill);
        }

        // 원 테두리의 자리 그라데이션: 0~100 자리마다 색이 고정되도록 잘게 나눠 긋는다(이음새가 보이지 않게 살짝 겹친다).
        void RingAlong(Painter2D p, Vector2 size, float from, float to)
        {
            var r = Mathf.Min(size.x, size.y) / 2f;
            var width = thickness - 2f * Inner;
            if (width <= 0f) return;
            var colors = Colors();
            const float Step = 1f / 120f;
            for (var a = from; a < to; a += Step)
            {
                var b = Mathf.Min(to, a + Step);
                Stroke(p, size / 2f, r - thickness / 2f, width, a, Mathf.Min(to, b + 0.002f), colors.Evaluate((a + b) / 2f));
            }
        }

        // 그라데이션 색 지점: 코드의 Gradient → stops → color · end-color 둘.
        Gradient Colors()
        {
            if (_custom != null) return _custom;
            if (_built != null) return _built;
            var points = new List<(float t, Color c)>();
            if (!string.IsNullOrWhiteSpace(_stops))
            {
                foreach (var part in _stops.Split(','))
                {
                    var pair = part.Split(':');
                    if (pair.Length == 2 && float.TryParse(pair[0].Trim(), NumberStyles.Float, CultureInfo.InvariantCulture, out var t) && ColorUtility.TryParseHtmlString(pair[1].Trim(), out var c))
                        points.Add((t, c));
                }
                if (points.Exists(point => point.t > 1f)) for (var i = 0; i < points.Count; i++) points[i] = (points[i].t / 100f, points[i].c);   // 0~100으로 적었으면
                points.Sort((x, y) => x.t.CompareTo(y.t));
                if (points.Count > 8) points.RemoveRange(8, points.Count - 8);   // Unity Gradient는 색 지점 8개까지
            }
            if (points.Count < 2) points = new List<(float, Color)> { (0f, Fill), (1f, FillEnd) };
            _built = new Gradient();
            _built.SetKeys(points.ConvertAll(point => new GradientColorKey(point.c, Mathf.Clamp01(point.t))).ToArray(),
                points.ConvertAll(point => new GradientAlphaKey(point.c.a, Mathf.Clamp01(point.t))).ToArray());
            return _built;
        }

        void DrawOver(MeshGenerationContext context)
        {
            if (!Ready(out var size)) return;
            var p = context.painter2D;
            if (_preview < 0f) Range(p, size, Mathf.Max(0f, _main + _preview), _main, _previewColor);
            if (_kind == Kind.Line && _segments >= 2)
            {
                var inner = Rect(Boxes(size)[0], Inner);
                p.strokeColor = _tickColor;
                p.lineWidth = 1f;
                for (var i = 1; i < _segments; i++)
                {
                    var x = Mathf.Round(Mathf.Lerp(inner.xMin, inner.xMax, i / (float)_segments)) + 0.5f;
                    p.BeginPath();
                    p.MoveTo(Shear(new Vector2(x, inner.yMin), size));
                    p.LineTo(Shear(new Vector2(x, inner.yMax), size));
                    p.Stroke();
                }
            }
        }

        bool Ready(out Vector2 size)
        {
            size = contentRect.size;
            return !float.IsNaN(size.x) && size.x > 0f && size.y > 0f;
        }

        float Inner => _frame == Frame.Inset ? inset : 0f;

        // 게이지의 from~to(0~1) 몫을 종류에 맞게 그린다.
        void Range(Painter2D p, Vector2 size, float from, float to, Color? color)
        {
            if (to - from <= 0.0005f) return;
            switch (_kind)
            {
                case Kind.Ring:
                {
                    var r = Mathf.Min(size.x, size.y) / 2f;
                    var width = thickness - 2f * Inner;
                    if (width > 0f) Stroke(p, size / 2f, r - thickness / 2f, width, from, to, color ?? Fill);
                    break;
                }
                case Kind.Orb:
                {
                    var r = Mathf.Min(size.x, size.y) / 2f - Inner;
                    if (r > 0f) Paint(p, Band(size / 2f, r, from, to), color);
                    break;
                }
                default:
                {
                    // 가로 몫(fromRight면 뒤집는다)을 도형마다 나눈다 — 도형 나열은 한 칸씩 차례로.
                    float a = _fromRight ? 1f - to : from, b = _fromRight ? 1f - from : to;
                    var boxes = Boxes(size);
                    for (var i = 0; i < boxes.Count; i++)
                    {
                        float u0 = i / (float)boxes.Count, u1 = (i + 1) / (float)boxes.Count;
                        float s = Mathf.Max(a, u0), e = Mathf.Min(b, u1);
                        if (e - s <= 0.0005f) continue;
                        var inner = Rect(boxes[i], Inner);
                        var x0 = Mathf.Lerp(inner.xMin, inner.xMax, (s - u0) / (u1 - u0));
                        var x1 = Mathf.Lerp(inner.xMin, inner.xMax, (e - u0) / (u1 - u0));
                        var clipped = ClipX(ClipX(ShapeOf(boxes[i], size, Inner, unshearedOnly: true), x0, true), x1, false);
                        if (clipped.Count < 3) continue;
                        if (_shape != Shape.Diamond) for (var k = 0; k < clipped.Count; k++) clipped[k] = Shear(clipped[k], size);
                        Paint(p, clipped, color);
                    }
                    break;
                }
            }
        }

        // 직선은 상자 하나, 도형 나열은 units개(가로로, 사이 unitGap). 상자는 기울이기 전(unsheared) 좌표.
        List<Rect> Boxes(Vector2 size)
        {
            var sheared = _shape == Shape.Slant ? Mathf.Min(slant, size.x * 0.25f) : 0f;
            var width = size.x - sheared;
            var list = new List<Rect>();
            if (_kind != Kind.Units) { list.Add(new Rect(0f, 0f, width, size.y)); return list; }
            var unit = (width - unitGap * (_units - 1)) / _units;
            for (var i = 0; i < _units; i++) list.Add(new Rect(i * (unit + unitGap), 0f, unit, size.y));
            return list;
        }

        static Rect Rect(Rect box, float d) => new Rect(box.x + d, box.y + d, Mathf.Max(0f, box.width - 2f * d), Mathf.Max(0f, box.height - 2f * d));

        // 상자를 d만큼 안으로 줄인 도형의 점들(시계 방향). 사각 · 평행사변형은 모서리를 radius만큼 둥글리고 기울인다, 마름모는 기울이지 않는다.
        List<Vector2> ShapeOf(Rect box, Vector2 size, float d, bool unshearedOnly = false)
        {
            var points = new List<Vector2>();
            if (_shape == Shape.Diamond)
            {
                var inner = Rect(box, d * 1.4f);
                var c = inner.center;
                var cut = Mathf.Min(radius, inner.width * 0.2f);
                AddBevel(points, new Vector2(c.x, inner.yMin), new Vector2(inner.xMax, c.y), new Vector2(inner.xMin, c.y), cut);
                AddBevel(points, new Vector2(inner.xMax, c.y), new Vector2(c.x, inner.yMax), new Vector2(c.x, inner.yMin), cut);
                AddBevel(points, new Vector2(c.x, inner.yMax), new Vector2(inner.xMin, c.y), new Vector2(inner.xMax, c.y), cut);
                AddBevel(points, new Vector2(inner.xMin, c.y), new Vector2(c.x, inner.yMin), new Vector2(c.x, inner.yMax), cut);
                return points;
            }
            var r = Rect(box, d);
            var round = Mathf.Min(radius, r.width / 2f, r.height / 2f);
            AddCorner(points, new Vector2(r.xMin + round, r.yMin + round), round, 180f);
            AddCorner(points, new Vector2(r.xMax - round, r.yMin + round), round, 270f);
            AddCorner(points, new Vector2(r.xMax - round, r.yMax - round), round, 0f);
            AddCorner(points, new Vector2(r.xMin + round, r.yMax - round), round, 90f);
            if (!unshearedOnly) for (var i = 0; i < points.Count; i++) points[i] = Shear(points[i], size);
            return points;
        }

        static void AddCorner(List<Vector2> points, Vector2 center, float r, float startDegrees)
        {
            if (r <= 0.01f) { points.Add(center); return; }
            for (var i = 0; i <= 3; i++)
            {
                var a = (startDegrees + i * 30f) * Mathf.Deg2Rad;
                points.Add(center + new Vector2(Mathf.Cos(a), Mathf.Sin(a)) * r);
            }
        }

        // 꼭짓점 corner를 이웃 점 쪽으로 cut만큼 잘라 낸 두 점(아주 약간의 둥글림 대신).
        static void AddBevel(List<Vector2> points, Vector2 corner, Vector2 next, Vector2 previous, float cut)
        {
            if (cut <= 0.01f) { points.Add(corner); return; }
            points.Add(corner + (previous - corner).normalized * cut);
            points.Add(corner + (next - corner).normalized * cut);
        }

        // 평행사변형: 아래 변을 기준으로 위로 갈수록 오른쪽으로 민다.
        Vector2 Shear(Vector2 point, Vector2 size)
        {
            if (_shape != Shape.Slant) return point;
            var sheared = Mathf.Min(slant, size.x * 0.25f);
            return new Vector2(point.x + sheared * (size.y - point.y) / size.y, point.y);
        }

        // 볼록 다각형을 세로선 x로 자른다(keepRight = 그 오른쪽을 남김).
        static List<Vector2> ClipX(List<Vector2> polygon, float x, bool keepRight)
        {
            var result = new List<Vector2>();
            for (var i = 0; i < polygon.Count; i++)
            {
                var a = polygon[i];
                var b = polygon[(i + 1) % polygon.Count];
                var aIn = keepRight ? a.x >= x : a.x <= x;
                var bIn = keepRight ? b.x >= x : b.x <= x;
                if (aIn) result.Add(a);
                if (aIn != bIn) result.Add(Vector2.Lerp(a, b, (x - a.x) / (b.x - a.x)));
            }
            return result;
        }

        // 원 안의 from~to 높이 띠(아래에서 위로 차오름).
        static List<Vector2> Band(Vector2 c, float r, float from, float to)
        {
            var bottom = c.y + r - from * 2f * r;
            var top = c.y + r - to * 2f * r;
            var points = new List<Vector2>();
            const int Steps = 24;
            for (var i = 0; i <= Steps; i++)
            {
                var y = Mathf.Lerp(top, bottom, i / (float)Steps);
                points.Add(new Vector2(c.x + HalfWidth(c, r, y), y));
            }
            for (var i = Steps; i >= 0; i--)
            {
                var y = Mathf.Lerp(top, bottom, i / (float)Steps);
                points.Add(new Vector2(c.x - HalfWidth(c, r, y), y));
            }
            return points;
        }

        static float HalfWidth(Vector2 c, float r, float y) => Mathf.Sqrt(Mathf.Max(0f, r * r - (y - c.y) * (y - c.y)));

        // 자리 그라데이션(직선 · 도형 나열 = 가로 0~100, 원형 패널 = 아래 0 → 위 100). 원 테두리는 RingAlong.
        void UseGradient(Painter2D p, Vector2 size)
        {
            if (_kind == Kind.Orb)
            {
                var r = Mathf.Min(size.x, size.y) / 2f - Inner;
                p.fillGradient = FillGradient.MakeLinearGradient(Colors(), new Vector2(size.x / 2f, size.y / 2f + r), new Vector2(size.x / 2f, size.y / 2f - r), AddressMode.Clamp);
                return;
            }
            var boxes = Boxes(size);
            float left = boxes[0].xMin + Inner, right = boxes[boxes.Count - 1].xMax - Inner;
            var start = new Vector2(_fromRight ? right : left, size.y / 2f);
            var end = new Vector2(_fromRight ? left : right, size.y / 2f);
            p.fillGradient = FillGradient.MakeLinearGradient(Colors(), start, end, AddressMode.Clamp);
        }

        // color가 null이면 칠하기 상태(그라데이션)를 그대로 쓴다.
        static void Paint(Painter2D p, List<Vector2> points, Color? color)
        {
            if (points.Count < 3) return;
            if (color.HasValue) p.fillColor = color.Value;
            p.BeginPath();
            p.MoveTo(points[0]);
            for (var i = 1; i < points.Count; i++) p.LineTo(points[i]);
            p.ClosePath();
            p.Fill();
        }

        static void Disc(Painter2D p, Vector2 c, float r, Color color)
        {
            if (r <= 0f) return;
            p.fillColor = color;
            p.BeginPath();
            p.Arc(c, r, Angle.Degrees(0f), Angle.Degrees(360f));
            p.ClosePath();
            p.Fill();
        }

        // 원 테두리의 from~to 몫(12시부터 시계 방향).
        static void Stroke(Painter2D p, Vector2 c, float r, float width, float from, float to, Color color)
        {
            if (r <= 0f || width <= 0f) return;
            p.strokeColor = color;
            p.lineWidth = width;
            p.lineCap = LineCap.Butt;
            p.BeginPath();
            p.Arc(c, r, Angle.Degrees(-90f + from * 360f), Angle.Degrees(-90f + to * 360f), ArcDirection.Clockwise);
            p.Stroke();
        }
    }
}
