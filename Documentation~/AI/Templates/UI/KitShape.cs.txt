// GBible UI 키트의 모양 칸(AI/Templates/UI) — 프로젝트의 Assets/MomoiProject/Code/Client/Kit/KitShape.cs로 복사한다(네임스페이스는 프로젝트 것).
// USS로 만들 수 없는 네모가 아닌 면을 그린다: 비스듬한 칸(탭·제목 막대) · 한쪽만 비스듬한 끝(꼬리표) · 모서리 깎기(카드·창).
// 둥근 모양(알약·원·아치)은 USS border-radius로 충분하다 — 이 칸은 비스듬한 것만 맡는다.
// 색은 USS 사용자 속성 --kit-fill(:hover 등 상태 선택자로 바꿀 수 있다). 결 텍스처는 그리지 않는다(작은 칸이라 색만).
// UXML: <MomoiProject.KitShape shape="Slant" slant="12" class="kit-tab"><ui:Label text="…" class="kit-tab-label" /></MomoiProject.KitShape>
using UnityEngine;
using UnityEngine.UIElements;

namespace MomoiProject
{
    [UxmlElement]
    public partial class KitShape : VisualElement
    {
        public enum Kind
        {
            Slant,      // 평행사변형(왼쪽 아래·오른쪽 위로 기움)
            SlantEnd,   // 왼쪽은 곧고 오른쪽 끝만 비스듬함
            CutCorner,  // 왼쪽 위·오른쪽 아래 모서리를 깎음
        }

        static readonly CustomStyleProperty<Color> FillProperty = new CustomStyleProperty<Color>("--kit-fill");

        [UxmlAttribute] public Kind shape { get; set; } = Kind.Slant;
        [UxmlAttribute] public float slant { get; set; } = 12f;

        Color _fill = new Color(0.141f, 0.145f, 0.161f);

        public KitShape()
        {
            generateVisualContent += Draw;
            RegisterCallback<CustomStyleResolvedEvent>(e =>
            {
                if (e.customStyle.TryGetValue(FillProperty, out var color)) _fill = color;
                MarkDirtyRepaint();
            });
        }

        void Draw(MeshGenerationContext context)
        {
            var w = layout.width;
            var h = layout.height;
            if (float.IsNaN(w) || w <= 0 || h <= 0 || _fill.a <= 0) return;
            var s = Mathf.Clamp(slant, 0, Mathf.Min(w, h));
            var painter = context.painter2D;
            painter.fillColor = _fill;
            painter.BeginPath();
            switch (shape)
            {
                case Kind.SlantEnd:
                    painter.MoveTo(new Vector2(0, 0));
                    painter.LineTo(new Vector2(w, 0));
                    painter.LineTo(new Vector2(w - s, h));
                    painter.LineTo(new Vector2(0, h));
                    break;
                case Kind.CutCorner:
                    painter.MoveTo(new Vector2(s, 0));
                    painter.LineTo(new Vector2(w, 0));
                    painter.LineTo(new Vector2(w, h - s));
                    painter.LineTo(new Vector2(w - s, h));
                    painter.LineTo(new Vector2(0, h));
                    painter.LineTo(new Vector2(0, s));
                    break;
                default:
                    painter.MoveTo(new Vector2(s, 0));
                    painter.LineTo(new Vector2(w, 0));
                    painter.LineTo(new Vector2(w - s, h));
                    painter.LineTo(new Vector2(0, h));
                    break;
            }
            painter.ClosePath();
            painter.Fill();
        }
    }
}
